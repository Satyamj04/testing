Bedrock

The frontend should only communicate with the backend API.

Example:

export async function createSecurityEvent(data) {
  const response = await fetch(
    `${API_BASE_URL}/security-event`,
    {
      method: "POST",
      headers: {
        "Content-Type": "application/json"
      },
      body: JSON.stringify(data)
    }
  );

  if (!response.ok) {
    throw new Error("Security event API failed");
  }

  return response.json();
}
